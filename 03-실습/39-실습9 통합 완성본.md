## 9-2. API 서버로 감싸기

사내 시스템에 붙이려면 HTTP API가 필요합니다.

```powershell

```

`C:\llm-lab\api.py`

```python
import time
import json
import datetime
import hashlib
from pathlib import Path

from fastapi import FastAPI, Header, HTTPException
from fastapi.responses import StreamingResponse
from pydantic import BaseModel

from config import get_client, CHAT_MODEL
from rag import retrieve, build_context, SYSTEM_PROMPT

app = FastAPI(title="사내 규정 안내 API")
llm = get_client()

LOG_PATH = Path("logs/qa.jsonl")
LOG_PATH.parent.mkdir(exist_ok=True)


class AskRequest(BaseModel):
    question: str
    top_k: int = 4
    stream: bool = False


def log_qa(user: str, question: str, answer: str, hits: list, latency_ms: int):
    with LOG_PATH.open("a", encoding="utf-8") as f:
        f.write(json.dumps({
            "ts": datetime.datetime.now().isoformat(),
            "user": hashlib.sha256(user.encode()).hexdigest()[:16],
            "question": question,
            "answer": answer,
            "sources": [h["source"] for h in hits],
            "top_score": hits[0]["score"] if hits else None,
            "latency_ms": latency_ms,
            "model": CHAT_MODEL,
        }, ensure_ascii=False) + "\n")


@app.get("/health")
def health():
    """운영 모니터링용. 로드밸런서가 이걸 봅니다."""
    try:
        llm.models.list()
        return {"status": "ok", "model": CHAT_MODEL}
    except Exception as e:
        raise HTTPException(503, f"LLM 서버 연결 실패: {e}")


@app.post("/ask")
def ask(req: AskRequest, x_user_id: str = Header(default="anonymous")):
    started = time.time()

    hits = [h for h in retrieve(req.question, req.top_k) if h["score"] >= 0.3]
    if not hits:
        answer = "관련된 사내 자료를 찾지 못했습니다. 담당 부서에 문의해주세요."
        log_qa(x_user_id, req.question, answer, [], int((time.time() - started) * 1000))
        return {"answer": answer, "sources": []}

    messages = [
        {"role": "system", "content": SYSTEM_PROMPT},
        {"role": "user",
         "content": f"[참고자료]\n{build_context(hits)}\n\n[질문]\n{req.question}"},
    ]

    if req.stream:
        def generate():
            collected = []
            for chunk in llm.chat.completions.create(
                model=CHAT_MODEL, messages=messages, temperature=0.1, stream=True
            ):
                delta = chunk.choices[0].delta.content
                if delta:
                    collected.append(delta)
                    yield f"data: {json.dumps({'delta': delta}, ensure_ascii=False)}\n\n"

            sources = [{"source": h["source"], "score": round(h["score"], 3)} for h in hits]
            yield f"data: {json.dumps({'sources': sources}, ensure_ascii=False)}\n\n"
            yield "data: [DONE]\n\n"

            log_qa(x_user_id, req.question, "".join(collected), hits,
                   int((time.time() - started) * 1000))

        return StreamingResponse(generate(), media_type="text/event-stream")

    resp = llm.chat.completions.create(model=CHAT_MODEL, messages=messages, temperature=0.1)
    answer = resp.choices[0].message.content

    log_qa(x_user_id, req.question, answer, hits, int((time.time() - started) * 1000))

    return {
        "answer": answer,
        "sources": [{"source": h["source"], "score": round(h["score"], 3)} for h in hits],
        "latency_ms": int((time.time() - started) * 1000),
    }
```

## 9-3. 품질 평가 스크립트

**모델이나 설정을 바꿀 때 좋아졌는지 나빠졌는지 알아야 합니다.**

```json
    "must_include": ["1년"],
    "must_not_include": []
  },
  {
    "question": "육아휴직 신청은 언제까지 해야 하나요?",
    "expected_source": "취업규칙.txt",
    "must_include": ["30일"],
    "must_not_include": []
  },
  {
    "question": "우리 회사 주식 배당금은 언제 지급되나요?",
    "expected_source": null,
    "must_include": ["확인되지 않"],
    "must_not_include": []
  }

```

`C:\llm-lab\eval.py`

```python
import json, time
from pathlib import Path
from rag import ask

cases = json.loads(Path("eval_set.json").read_text(encoding="utf-8"))

passed = 0
results = []

for c in cases:
    started = time.time()
    answer, hits = ask(c["question"])
    latency = int((time.time() - started) * 1000)

    checks = []
    for kw in c["must_include"]:
        checks.append(("포함", kw, kw in answer))
    for kw in c["must_not_include"]:
        checks.append(("제외", kw, kw not in answer))

    if c["expected_source"]:
        found = any(h["source"] == c["expected_source"] for h in hits)
        checks.append(("출처", c["expected_source"], found))

    ok = all(r for _, _, r in checks)
    passed += ok

    results.append({"q": c["question"], "ok": ok, "latency": latency,
                    "checks": checks, "answer": answer})

print(f"\n{'='*60}")
print(f"통과: {passed}/{len(cases)}  ({passed/len(cases)*100:.0f}%)")
print(f"평균 응답시간: {sum(r['latency'] for r in results)//len(results)}ms")
print(f"{'='*60}\n")

for r in results:
    mark = "PASS" if r["ok"] else "FAIL"
    print(f"[{mark}] {r['q']}  ({r['latency']}ms)")
    if not r["ok"]:
        for kind, kw, res in r["checks"]:
            if not res:
                print(f"       {kind} 실패: {kw}")
        print(f"       답변: {r['answer'][:120]}...")
```

**골든셋은 최소 30개, 가능하면 100개**를 만드세요. 현업 담당자와 같이 만드는 게 좋습니다. 진짜 자주 들어오는 질문을 그들이 알고 있으니까요.

**세 번째 케이스(답이 없어야 하는 질문)를 꼭 넣으세요.** 이게 환각 방어율을 재는 지표입니다. 전체 케이스의 20~30%를 이런 걸로 채우는 걸 권합니다.

## 9-5. 시연 시나리오

경영진이나 심의위원회 앞에서 보여줄 순서입니다.

1. **인터넷 차단 상태를 먼저 보여줍니다.** 랜선을 뽑거나 `ping 8.8.8.8` 실패를 화면에 띄웁니다. 이게 첫 번째 관문입니다
2. 사내 문서 기반 질문 → 정확한 답 + 출처 표시
3. **자료에 없는 질문 → "확인되지 않습니다"** ← 이게 신뢰를 얻는 핵심 장면입니다
4. 개인 정보 조회 (권한 있는 사번) → 정상 조회
5. **권한 없는 사번 조회 시도 → 거부되고 로그에 남음** ← 통제가 된다는 증거
6. 감사 로그 파일을 열어서 보여줍니다

3번과 5번이 승인 여부를 가릅니다. "잘 맞춘다"보다 **"틀릴 때 틀린다고 말하고, 안 되는 건 막힌다"**가 금융권에서 훨씬 중요합니다.
